One partial rather than four copies of the same markup, because the day, the week and the calculator all answer the same question and should not answer it in three different layouts. #} {% macro tiles(totals, target=none, pct=none, remaining=none) %}
{% for key in macro_keys %}
{{ macro_labels[key] }} {{ totals[key] | macro(key) }} {{ macro_units[key] }}{% if target and target[key] %} of {{ target[key] | macro(key) }}{% endif %} {% if pct and pct[key] is not none %} {# A bar rather than a number, because "how much of it is gone" is the question and a phone reads a length faster than a percentage. Over target is drawn full and coloured rather than overflowing. #} {# "146 over" rather than "-146 left", which is a sentence nobody says. The sign is already carried by the word, so the number is shown unsigned. #} {{- (pct[key] * 100) | round | int }}% {%- if remaining %} · {%- if remaining[key] < 0 %} {{ (0 - remaining[key]) | macro(key) }} over {%- else %} {{ remaining[key] | macro(key) }} left{% endif %}{% endif %} {% endif %}
{% endfor %}
